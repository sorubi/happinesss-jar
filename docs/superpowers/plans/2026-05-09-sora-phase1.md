```typescript
import Dexie, { type Table } from 'dexie'
import type { Orb } from '../types'

class OrbDatabase extends Dexie {
  orbs!: Table<Orb, string>

  constructor() {
    super('SoraDB')
    this.version(1).stores({
      orbs: 'id, year, month, createdAt',
    })
  }
}

export const orbDb = new OrbDatabase()

export async function addOrb(data: Omit<Orb, 'id'>): Promise<string> {
  const id = crypto.randomUUID()
  await orbDb.orbs.add({ ...data, id })
  return id
}

export async function getOrbsByMonth(year: number, month: number): Promise<Orb[]> {
  return orbDb.orbs.where({ year, month }).toArray()
}

export async function deleteOrb(id: string): Promise<void> {
  await orbDb.orbs.delete(id)
}

export async function updateOrb(id: string, changes: Partial<Pick<Orb, 'text' | 'color' | 'emotion'>>): Promise<void> {
  await orbDb.orbs.update(id, changes)
}
```

```typescript
import { create } from 'zustand'
import type { Orb } from '../types'

interface OrbStore {
  activeMonth: number
  activeYear: number
  orbs: Orb[]
  phantomText: string | null
  setActiveMonth: (month: number) => void
  setActiveYear: (year: number) => void
  setOrbs: (orbs: Orb[]) => void
  addOrbToCache: (orb: Orb) => void
  removeOrbFromCache: (id: string) => void
  setPhantomText: (text: string | null) => void
}

const now = new Date()

export const useOrbStore = create<OrbStore>((set) => ({
  activeMonth: now.getMonth(),
  activeYear: now.getFullYear(),
  orbs: [],
  phantomText: null,
  setActiveMonth: (month) => set({ activeMonth: month }),
  setActiveYear: (year) => set({ activeYear: year }),
  setOrbs: (orbs) => set({ orbs }),
  addOrbToCache: (orb) => set((s) => ({ orbs: [...s.orbs, orb] })),
  removeOrbFromCache: (id) => set((s) => ({ orbs: s.orbs.filter(o => o.id !== id) })),
  setPhantomText: (text) => set({ phantomText: text }),
}))
```

```typescript
import type { EmotionCategory } from '../types'

const KEYWORDS: Record<EmotionCategory, string[]> = {
  passion: ['도전', '성공', '노력', '인정', '완료', '달성', '목표', '열심', '프로젝트', '결과', '이겼'],
  peace: ['힐링', '산책', '여유', '휴식', '쉬었', '잠', '평온', '조용', '차분', '명상'],
  joy: ['맛집', '친구', '웃음', '재밌', '즐거', '행복', '신났', '좋았', '맛있', '기쁘'],
  gratitude: ['감사', '고마', '사랑', '가족', '엄마', '아빠', '선물', '배려', '따뜻', '응원'],
  excitement: ['설레', '기대', '여행', '새로', '시작', '만남', '처음', '계획', '두근'],
}

export function detectEmotion(text: string): EmotionCategory {
  const scores: Record<EmotionCategory, number> = {
    passion: 0, peace: 0, joy: 0, gratitude: 0, excitement: 0,
  }

  for (const [emotion, keywords] of Object.entries(KEYWORDS) as [EmotionCategory, string[]][]) {
    for (const kw of keywords) {
      if (text.includes(kw)) scores[emotion]++
    }
  }

  const top = (Object.entries(scores) as [EmotionCategory, number][])
    .sort((a, b) => b[1] - a[1])[0]

  return top[1] > 0 ? top[0] : 'joy'
}

export function getOrbColor(emotion: EmotionCategory): string {
  const MAP: Record<EmotionCategory, string> = {
    passion: '#FF6B9D',
    peace: '#4ECDC4',
    joy: '#FFE66D',
    gratitude: '#A259FF',
    excitement: '#FF8B94',
  }
  return MAP[emotion]
}

export function getOrbSize(text: string): number {
  const MIN = 30, MAX = 60
  const ratio = Math.min(1, text.length / 100)
  return Math.round(MIN + (MAX - MIN) * ratio)
}
```

```typescript
import type { JarGeometry, PhysicsOrb } from '../../types'

// ─── 헬퍼 ────────────────────────────────────────────────────────────────────

export function hexToRgb(hex: string): { r: number; g: number; b: number } {
  const n = parseInt(hex.slice(1), 16)
  return { r: (n >> 16) & 255, g: (n >> 8) & 255, b: n & 255 }
}

export function buildJarPath(ctx: CanvasRenderingContext2D, g: JarGeometry, inset = 0): void {
  const { cx, cy, bodyW, bodyH, neckW, neckH } = g
  const hw = bodyW / 2 - inset
  const hh = bodyH / 2
  const r = 26 - inset  // bottom corner radius

  ctx.beginPath()
  // Neck → top-left body
  ctx.moveTo(cx - neckW / 2, cy - hh)
  // Body: left side
  ctx.lineTo(cx - hw, cy - hh + 10)
  ctx.lineTo(cx - hw, cy + hh - r)
  ctx.quadraticCurveTo(cx - hw, cy + hh, cx - hw + r, cy + hh)
  // Body: bottom
  ctx.lineTo(cx + hw - r, cy + hh)
  ctx.quadraticCurveTo(cx + hw, cy + hh, cx + hw, cy + hh - r)
  // Body: right side
  ctx.lineTo(cx + hw, cy - hh + 10)
  ctx.lineTo(cx + neckW / 2, cy - hh)
  ctx.closePath()
}

export function buildNeckPath(ctx: CanvasRenderingContext2D, g: JarGeometry): void {
  const { cx, cy, bodyW, bodyH, neckW, neckH } = g
  ctx.beginPath()
  ctx.moveTo(cx - neckW / 2, cy - bodyH / 2)
  ctx.lineTo(cx - neckW / 2, cy - bodyH / 2 - neckH)
  ctx.lineTo(cx + neckW / 2, cy - bodyH / 2 - neckH)
  ctx.lineTo(cx + neckW / 2, cy - bodyH / 2)
  ctx.closePath()
}

// ─── 배경 ─────────────────────────────────────────────────────────────────────

export function drawBackground(ctx: CanvasRenderingContext2D, W: number, H: number): void {
  const grad = ctx.createLinearGradient(0, 0, 0, H)
  grad.addColorStop(0, '#0F172A')
  grad.addColorStop(1, '#1E1B4B')
  ctx.fillStyle = grad
  ctx.fillRect(0, 0, W, H)
}

// ─── 유리병 뒷면 (구슬 뒤에 그려짐) ─────────────────────────────────────────

export function drawJarBack(ctx: CanvasRenderingContext2D, g: JarGeometry): void {
  ctx.save()

  // Body fill (매우 투명한 유리)
  buildJarPath(ctx, g)
  ctx.fillStyle = 'rgba(80, 100, 180, 0.04)'
  ctx.fill()

  // Neck fill
  buildNeckPath(ctx, g)
  ctx.fillStyle = 'rgba(80, 100, 180, 0.04)'
  ctx.fill()

  ctx.restore()
}

// ─── 구슬 글로우 (Additive Blending — 이 함수를 lighter 모드에서 호출할 것) ──

export function drawOrbGlow(ctx: CanvasRenderingContext2D, orb: PhysicsOrb): void {
  const { x, y, radius, color } = orb
  const { r, g, b } = hexToRgb(color)

  // 3단계 글로우: 바깥부터 안으로, 각각 다른 크기와 강도
  const layers = [
    { mult: 5.5, alpha: 0.05 },
    { mult: 3.2, alpha: 0.09 },
    { mult: 1.9, alpha: 0.16 },
  ]

  for (const { mult, alpha } of layers) {
    const gr = radius * mult
    const grad = ctx.createRadialGradient(x, y, 0, x, y, gr)
    grad.addColorStop(0,   `rgba(${r},${g},${b},${alpha})`)
    grad.addColorStop(0.4, `rgba(${r},${g},${b},${alpha * 0.5})`)
    grad.addColorStop(1,   `rgba(${r},${g},${b},0)`)
    ctx.beginPath()
    ctx.arc(x, y, gr, 0, Math.PI * 2)
    ctx.fillStyle = grad
    ctx.fill()
  }
}

// ─── 구슬 본체 (white-hot 중심, 소프트 엣지) ──────────────────────────────────

export function drawOrbBody(ctx: CanvasRenderingContext2D, orb: PhysicsOrb): void {
  const { x, y, radius, color } = orb
  const { r, g, b } = hexToRgb(color)

  // 소프트 엣지: 반지름보다 약간 크게 그려서 경계를 투명하게 페이드
  const R = radius * 1.1

  const grad = ctx.createRadialGradient(
    x - radius * 0.12, y - radius * 0.18, 0,
    x, y, R
  )
  grad.addColorStop(0,    'rgba(255,255,255,0.92)')         // white-hot 중심
  grad.addColorStop(0.18, `rgba(${r},${g},${b},0.95)`)     // 본 색상
  grad.addColorStop(0.60, `rgba(${r},${g},${b},0.82)`)
  grad.addColorStop(0.82, `rgba(${r},${g},${b},0.40)`)
  grad.addColorStop(1,    `rgba(${r},${g},${b},0.0)`)       // 소프트 엣지

  ctx.beginPath()
  ctx.arc(x, y, R, 0, Math.PI * 2)
  ctx.fillStyle = grad
  ctx.fill()
}

// ─── 구슬들이 내뿜는 앰비언트 인테리어 라이트 ────────────────────────────────

export function drawAmbientInterior(
  ctx: CanvasRenderingContext2D,
  orbs: PhysicsOrb[],
  g: JarGeometry,
): void {
  if (orbs.length === 0) return

  // 지배적인 색상 (가장 많이 등장한 색)
  const colorCount: Record<string, number> = {}
  orbs.forEach(o => { colorCount[o.color] = (colorCount[o.color] ?? 0) + 1 })
  const dominant = Object.entries(colorCount).sort((a, b) => b[1] - a[1])[0][0]
  const { r, rv, b: bv } = hexToRgb(dominant) as { r: number; rv?: never; b: number } & ReturnType<typeof hexToRgb>

  const cx = g.cx
  const bottomY = g.cy + g.bodyH / 2

  // 병 바닥에서 올라오는 따뜻한 빛
  const ambGrad = ctx.createRadialGradient(cx, bottomY, 0, cx, bottomY, g.bodyW * 0.7)
  ambGrad.addColorStop(0,   `rgba(${r},${hexToRgb(dominant).g},${bv},0.22)`)
  ambGrad.addColorStop(0.5, `rgba(${r},${hexToRgb(dominant).g},${bv},0.08)`)
  ambGrad.addColorStop(1,   `rgba(${r},${hexToRgb(dominant).g},${bv},0)`)
  ctx.fillStyle = ambGrad
  ctx.fillRect(cx - g.bodyW, bottomY - g.bodyH, g.bodyW * 2, g.bodyH)
}

// ─── 유리병 앞면 (구슬 위에 그려짐) ─────────────────────────────────────────

export function drawJarFront(ctx: CanvasRenderingContext2D, g: JarGeometry): void {
  const { cx, cy, bodyW, bodyH, neckW, neckH } = g
  ctx.save()

  // Body outline
  buildJarPath(ctx, g)
  ctx.strokeStyle = 'rgba(148, 163, 184, 0.26)'
  ctx.lineWidth = 1.5
  ctx.stroke()

  // Neck outline
  buildNeckPath(ctx, g)
  ctx.strokeStyle = 'rgba(148, 163, 184, 0.26)'
  ctx.lineWidth = 1.5
  ctx.stroke()

  // Neck top rim (두꺼운 선)
  ctx.beginPath()
  ctx.moveTo(cx - neckW / 2 - 4, cy - bodyH / 2 - neckH)
  ctx.lineTo(cx + neckW / 2 + 4, cy - bodyH / 2 - neckH)
  ctx.strokeStyle = 'rgba(148, 163, 184, 0.35)'
  ctx.lineWidth = 2.5
  ctx.stroke()

  // 왼쪽 엣지 유리 하이라이트
  ctx.beginPath()
  ctx.moveTo(cx - bodyW / 2 + 7, cy - bodyH / 2 + 18)
  ctx.lineTo(cx - bodyW / 2 + 7, cy + bodyH / 2 - 32)
  ctx.strokeStyle = 'rgba(255, 255, 255, 0.07)'
  ctx.lineWidth = 4
  ctx.lineCap = 'round'
  ctx.stroke()

  ctx.restore()
}

// ─── 전체 씬 렌더 (이 함수만 외부에서 호출) ──────────────────────────────────

export function renderJarScene(
  ctx: CanvasRenderingContext2D,
  W: number,
  H: number,
  orbs: PhysicsOrb[],
  g: JarGeometry,
): void {
  ctx.clearRect(0, 0, W, H)

  // 1. 배경
  drawBackground(ctx, W, H)

  // 2. 병 뒷면
  drawJarBack(ctx, g)

  // 3. 클리핑 — 구슬은 병 안에만
  ctx.save()
  buildJarPath(ctx, g, 2)
  ctx.clip()

  // 4a. 구슬 글로우 — Additive Blending (핵심!)
  ctx.save()
  ctx.globalCompositeOperation = 'lighter'
  orbs.forEach(o => drawOrbGlow(ctx, o))
  ctx.restore()

  // 4b. 구슬 본체
  orbs.forEach(o => drawOrbBody(ctx, o))

  // 4c. 앰비언트 인테리어 라이트
  drawAmbientInterior(ctx, orbs, g)

  ctx.restore() // 클리핑 해제

  // 5. 병 앞면 (구슬 위에)
  drawJarFront(ctx, g)
}
```

```typescript
import { useEffect, useRef, useCallback } from 'react'
import Matter from 'matter-js'
import type { JarGeometry, PhysicsOrb } from '../../types'

const { Engine, Bodies, Body, Composite, Events } = Matter

interface UsePhysicsOptions {
  geometry: JarGeometry
  onOrbsUpdate: (orbs: PhysicsOrb[]) => void
}

interface OrbBody {
  body: Matter.Body
  id: string
  color: string
  radius: number
}

export function usePhysics({ geometry: g, onOrbsUpdate }: UsePhysicsOptions) {
  const engineRef = useRef<Matter.Engine | null>(null)
  const orbBodiesRef = useRef<OrbBody[]>([])
  const rafRef = useRef<number>(0)

  // 엔진 초기화
  useEffect(() => {
    const engine = Engine.create({ gravity: { y: 2.5 } })
    engineRef.current = engine

    // 병 물리 벽 (보이지 않음)
    const T = 12  // 벽 두께
    const walls = [
      // 바닥
      Bodies.rectangle(g.cx, g.cy + g.bodyH / 2 + T / 2, g.bodyW + T * 2, T, { isStatic: true, label: 'floor' }),
      // 왼쪽 벽
      Bodies.rectangle(g.cx - g.bodyW / 2 - T / 2, g.cy, T, g.bodyH + g.neckH, { isStatic: true, label: 'lwall' }),
      // 오른쪽 벽
      Bodies.rectangle(g.cx + g.bodyW / 2 + T / 2, g.cy, T, g.bodyH + g.neckH, { isStatic: true, label: 'rwall' }),
    ]
    Composite.add(engine.world, walls)

    // 애니메이션 루프
    let lastTime = performance.now()
    const loop = (now: number) => {
      const delta = now - lastTime
      lastTime = now
      Engine.update(engine, Math.min(delta, 32))

      const physicsOrbs: PhysicsOrb[] = orbBodiesRef.current.map(({ body, id, color, radius }) => ({
        id,
        color,
        radius,
        x: body.position.x,
        y: body.position.y,
      }))
      onOrbsUpdate(physicsOrbs)

      rafRef.current = requestAnimationFrame(loop)
    }
    rafRef.current = requestAnimationFrame(loop)

    return () => {
      cancelAnimationFrame(rafRef.current)
      Engine.clear(engine)
    }
  }, [g.cx, g.cy, g.bodyW, g.bodyH, g.neckW, g.neckH])

  // 구슬 추가
  const addOrbToPhysics = useCallback((id: string, color: string, radius: number) => {
    const engine = engineRef.current
    if (!engine) return

    // 목 입구 중앙에서 약간 랜덤한 위치로 드롭
    const x = g.cx + (Math.random() - 0.5) * g.neckW * 0.5
    const y = g.cy - g.bodyH / 2 - g.neckH + radius

    const body = Bodies.circle(x, y, radius, {
      restitution: 0.42,
      friction: 0.08,
      frictionAir: 0.014,
      density: 0.002,
    })
    Composite.add(engine.world, body)
    orbBodiesRef.current.push({ body, id, color, radius })
  }, [g])

  // 흔들기 (Shake to Mix)
  const shake = useCallback(() => {
    orbBodiesRef.current.forEach(({ body }) => {
      Body.applyForce(body, body.position, {
        x: (Math.random() - 0.5) * 0.14,
        y: -(0.08 + Math.random() * 0.12),
      })
    })
  }, [])

  // 특정 구슬 제거
  const removeOrbFromPhysics = useCallback((id: string) => {
    const engine = engineRef.current
    if (!engine) return
    const idx = orbBodiesRef.current.findIndex(o => o.id === id)
    if (idx === -1) return
    Composite.remove(engine.world, orbBodiesRef.current[idx].body)
    orbBodiesRef.current.splice(idx, 1)
  }, [])

  return { addOrbToPhysics, shake, removeOrbFromPhysics }
}
```

```typescript
import { useRef, useEffect, useState, forwardRef, useImperativeHandle } from 'react'
import type { Orb, JarGeometry, PhysicsOrb } from '../../types'
import { renderJarScene } from './jarRenderer'
import { usePhysics } from './usePhysics'
import './JarCanvas.css'

interface JarCanvasProps {
  orbs: Orb[]
  width: number
  height: number
  onOrbTap?: (orb: Orb) => void
}

export interface JarCanvasRef {
  shake: () => void
}

// 캔버스 크기에서 JarGeometry 계산
function computeGeometry(W: number, H: number): JarGeometry {
  return {
    cx: W / 2,
    cy: H * 0.54,
    bodyW: W * 0.56,
    bodyH: H * 0.46,
    neckW: W * 0.22,
    neckH: H * 0.07,
  }
}

const JarCanvas = forwardRef<JarCanvasRef, JarCanvasProps>(
  ({ orbs, width, height, onOrbTap }, ref) => {
    const canvasRef = useRef<HTMLCanvasElement>(null)
    const [physicsOrbs, setPhysicsOrbs] = useState<PhysicsOrb[]>([])
    const geometry = computeGeometry(width, height)
    const prevOrbIdsRef = useRef<Set<string>>(new Set())

    const { addOrbToPhysics, shake, removeOrbFromPhysics } = usePhysics({
      geometry,
      onOrbsUpdate: setPhysicsOrbs,
    })

    // Shake ref 노출
    useImperativeHandle(ref, () => ({ shake }), [shake])

    // Orbs 변경 감지 → 물리 엔진에 추가/제거
    useEffect(() => {
      const newIds = new Set(orbs.map(o => o.id))

      // 추가된 구슬
      for (const orb of orbs) {
        if (!prevOrbIdsRef.current.has(orb.id)) {
          addOrbToPhysics(orb.id, orb.color, orb.size)
        }
      }

      // 제거된 구슬
      for (const id of prevOrbIdsRef.current) {
        if (!newIds.has(id)) {
          removeOrbFromPhysics(id)
        }
      }

      prevOrbIdsRef.current = newIds
    }, [orbs, addOrbToPhysics, removeOrbFromPhysics])

    // 렌더링 루프
    useEffect(() => {
      const canvas = canvasRef.current
      if (!canvas) return
      const ctx = canvas.getContext('2d')
      if (!ctx) return
      renderJarScene(ctx, width, height, physicsOrbs, geometry)
    }, [physicsOrbs, width, height, geometry])

    // 탭으로 구슬 선택
    function handleClick(e: React.MouseEvent<HTMLCanvasElement>) {
      if (!onOrbTap) return
      const rect = canvasRef.current!.getBoundingClientRect()
      const mx = (e.clientX - rect.left) * (width / rect.width)
      const my = (e.clientY - rect.top) * (height / rect.height)

      // 클릭 위치와 가장 가까운 구슬 찾기
      for (const po of physicsOrbs) {
        const dist = Math.sqrt((mx - po.x) ** 2 + (my - po.y) ** 2)
        if (dist <= po.radius * 1.3) {
          const orb = orbs.find(o => o.id === po.id)
          if (orb) onOrbTap(orb)
          return
        }
      }
    }

    return (
      <div className="jar-canvas-wrap">
        <canvas
          ref={canvasRef}
          width={width}
          height={height}
          onClick={handleClick}
        />
      </div>
    )
  }
)

JarCanvas.displayName = 'JarCanvas'
export default JarCanvas
```

```typescript
import { gsap } from 'gsap'

// 텍스트 → 구슬 Materialization 애니메이션
// canvas에 텍스트 입자 → 구슬로 수렴하는 효과를 그림
export function playMaterializeAnim(
  canvas: HTMLCanvasElement,
  color: string,
  onComplete: () => void,
): void {
  const ctx = canvas.getContext('2d')
  if (!ctx) { onComplete(); return }

  const W = canvas.width, H = canvas.height
  const cx = W / 2, cy = H / 2

  // 파티클: 텍스트 위치에서 시작 → 중심으로 수렴
  const particles = Array.from({ length: 40 }, () => ({
    x: cx + (Math.random() - 0.5) * W * 0.6,
    y: cy + (Math.random() - 0.5) * H * 0.3,
    tx: cx,
    ty: cy,
    alpha: 1,
    size: 2 + Math.random() * 3,
  }))

  const state = { progress: 0 }

  gsap.to(state, {
    progress: 1,
    duration: 1.4,
    ease: 'power2.inOut',
    onUpdate() {
      ctx.clearRect(0, 0, W, H)
      const p = state.progress

      particles.forEach(pt => {
        const x = pt.x + (pt.tx - pt.x) * p
        const y = pt.y + (pt.ty - pt.y) * p
        ctx.save()
        ctx.globalAlpha = pt.alpha * (1 - p * 0.5)
        ctx.fillStyle = color
        ctx.shadowBlur = 6
        ctx.shadowColor = color
        ctx.beginPath()
        ctx.arc(x, y, pt.size * (1 - p * 0.3), 0, Math.PI * 2)
        ctx.fill()
        ctx.restore()
      })

      // 구슬 형성 (progress 0.6 이후)
      if (p > 0.6) {
        const orbProgress = (p - 0.6) / 0.4
        const radius = 30 * orbProgress
        const { r, g, b } = hexToRgb(color)

        ctx.save()
        ctx.globalCompositeOperation = 'lighter'
        const glowGrad = ctx.createRadialGradient(cx, cy, 0, cx, cy, radius * 3)
        glowGrad.addColorStop(0, `rgba(${r},${g},${b},${0.3 * orbProgress})`)
        glowGrad.addColorStop(1, `rgba(${r},${g},${b},0)`)
        ctx.beginPath()
        ctx.arc(cx, cy, radius * 3, 0, Math.PI * 2)
        ctx.fillStyle = glowGrad
        ctx.fill()
        ctx.restore()

        const bodyGrad = ctx.createRadialGradient(cx - radius * 0.15, cy - radius * 0.2, 0, cx, cy, radius * 1.1)
        bodyGrad.addColorStop(0, 'rgba(255,255,255,0.9)')
        bodyGrad.addColorStop(0.2, `rgba(${r},${g},${b},0.95)`)
        bodyGrad.addColorStop(1, `rgba(${r},${g},${b},0)`)
        ctx.beginPath()
        ctx.arc(cx, cy, radius * 1.1, 0, Math.PI * 2)
        ctx.fillStyle = bodyGrad
        ctx.fill()
      }
    },
    onComplete,
  })
}

function hexToRgb(hex: string) {
  const n = parseInt(hex.slice(1), 16)
  return { r: (n >> 16) & 255, g: (n >> 8) & 255, b: n & 255 }
}
```

```typescript
import { useEffect, useRef, useCallback } from 'react'

interface UseShakeOptions {
  threshold?: number   // 흔들기 감지 임계값 (m/s², 기본 25)
  cooldown?: number    // 연속 감지 방지 쿨다운 (ms, 기본 800)
  onShake: () => void
}

export function useShake({ threshold = 25, cooldown = 800, onShake }: UseShakeOptions) {
  const lastShakeTime = useRef(0)
  const lastAccel = useRef({ x: 0, y: 0, z: 0 })

  const handleMotion = useCallback((event: DeviceMotionEvent) => {
    const acc = event.accelerationIncludingGravity
    if (!acc) return

    const { x = 0, y = 0, z = 0 } = acc
    const prev = lastAccel.current

    const delta = Math.sqrt(
      (x - prev.x) ** 2 +
      (y - prev.y) ** 2 +
      (z - prev.z) ** 2
    )

    lastAccel.current = { x: x ?? 0, y: y ?? 0, z: z ?? 0 }

    if (delta > threshold) {
      const now = Date.now()
      if (now - lastShakeTime.current > cooldown) {
        lastShakeTime.current = now
        onShake()
      }
    }
  }, [threshold, cooldown, onShake])

  useEffect(() => {
    // iOS 13+ 퍼미션 요청
    const setup = async () => {
      if (typeof (DeviceMotionEvent as unknown as { requestPermission?: () => Promise<string> }).requestPermission === 'function') {
        try {
          const perm = await (DeviceMotionEvent as unknown as { requestPermission: () => Promise<string> }).requestPermission()
          if (perm !== 'granted') return
        } catch {
          return
        }
      }
      window.addEventListener('devicemotion', handleMotion)
    }
    setup()
    return () => window.removeEventListener('devicemotion', handleMotion)
  }, [handleMotion])
}
```

```typescript
  const { activeMonth, activeYear } = useOrbStore()

  const now = new Date()
  const isCurrent = activeMonth === now.getMonth() && activeYear === now.getFullYear()

  return (

        <>
          <MainShelf onJarSelect={() => setView('detail')} />

      {view === 'detail' && (
        <JarDetail
          month={activeMonth}
          year={activeYear}
          onBack={() => setView('shelf')}
          onCapture={() => setShowCapture(true)}
        />
      )}

      {showCapture && (
        <CaptureModal onClose={() => setShowCapture(false)} />
      )}
    </>
  )
}
```
